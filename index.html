SQL Schema — انسخه ونفّذه في SQL Editor:
create table profiles ( id uuid references auth.users primary key, name text, avatar_url text, status_text text, bio text, banner_url text, visible boolean default true, created_at timestamptz default now() ); create table rooms ( id uuid default gen_random_uuid() primary key, name text not null, description text, room_type text default 'public', invite_code text, created_by uuid references profiles(id), created_at timestamptz default now() ); create table room_members ( room_id uuid references rooms(id) on delete cascade, user_id uuid references profiles(id), joined_at timestamptz default now(), primary key(room_id, user_id) ); create table messages ( id uuid default gen_random_uuid() primary key, room_id uuid references rooms(id) on delete cascade, user_id uuid references profiles(id), content text, image_url text, reply_to uuid references messages(id), created_at timestamptz default now() ); create table reactions ( id uuid default gen_random_uuid() primary key, msg_id uuid references messages(id) on delete cascade, user_id uuid references profiles(id), emoji text, unique(msg_id, user_id, emoji) ); create table typing_status ( room_id uuid references rooms(id) on delete cascade, user_id uuid references profiles(id), updated_at timestamptz default now(), primary key(room_id, user_id) ); create table friendships ( id uuid default gen_random_uuid() primary key, requester_id uuid references profiles(id) on delete cascade, receiver_id uuid references profiles(id) on delete cascade, status text default 'pending', created_at timestamptz default now(), unique(requester_id, receiver_id) ); -- RLS alter table profiles enable row level security; alter table rooms enable row level security; alter table room_members enable row level security; alter table messages enable row level security; alter table reactions enable row level security; alter table typing_status enable row level security; alter table friendships enable row level security; create policy "p_sel" on profiles for select using (true); create policy "p_ins" on profiles for insert with check (auth.uid()=id); create policy "p_upd" on profiles for update using (auth.uid()=id); create policy "r_sel" on rooms for select using (true); create policy "r_ins" on rooms for insert with check (auth.uid()=created_by); create policy "r_del" on rooms for delete using (auth.uid()=created_by); create policy "rm_sel" on room_members for select using (true); create policy "rm_ins" on room_members for insert with check (auth.uid()=user_id); create policy "m_sel" on messages for select using (true); create policy "m_ins" on messages for insert with check (auth.uid()=user_id); create policy "m_del" on messages for delete using (auth.uid()=user_id); create policy "re_sel" on reactions for select using (true); create policy "re_ins" on reactions for insert with check (auth.uid()=user_id); create policy "re_del" on reactions for delete using (auth.uid()=user_id); create policy "ty_sel" on typing_status for select using (true); create policy "ty_all" on typing_status for all using (auth.uid()=user_id); create policy "fr_sel" on friendships for select using (true); create policy "fr_ins" on friendships for insert with check (auth.uid()=requester_id); create policy "fr_upd" on friendships for update using (auth.uid()=receiver_id); create policy "fr_del" on friendships for delete using (auth.uid()=requester_id or auth.uid()=receiver_id); create policy "storage_sel" on storage.objects for select using (bucket_id='chat-images'); create policy "storage_ins" on storage.objects for insert with check (bucket_id='chat-images' and auth.uid() is not null); create policy "storage_del" on storage.objects for delete using (bucket_id='chat-images' and auth.uid()=owner);
